v_xc_LDA

Compute the exchange-correlation potential using the local-density approximation.

Syntax:

[out] = v_xc_LDA(mass,perm,rho,opt)

Description:

 Compute the exchange-correlation potential using the local-density approximation. 
 If the inputs are left empty, return a function for generating the exchange-correlation
 potential: out = func(mass,perm,dens). If the inputs "mass", "perm", and "dens"
  are given numeric values, return the numeric exchange-correlation potential.

0002 % Compute the exchange-correlation potential using the local-density approximation.
0003 % If the inputs are left empty, return a function for generating the exchange-correlation
0004 % potential: out = func(mass,perm,dens). If the inputs "mass", "perm", and "dens"
0005 %  are given numeric values, return the numeric exchange-correlation potential.
0006 arguments
0007     mass (:,1) double {mustBeFinite,mustBeReal,mustBeNonzero} = [];
0008     perm (:,1) double {mustBeFinite,mustBeReal,mustBeNonzero} = [];
0009     rho (:,1) double {mustBeFinite,mustBeReal} = [];
0010     opt.method {mustBeMember(opt.method,[1,2])} = 2;
0011 end
0012 
0013 % There are two sets of empirical values for this formulation of the LDA exchange-correlation
0014 % energy, found in units of Rydberg in the papers below
0015 switch opt.method
0016     case 1
0017         % L. Hedin and B. I. Lundqvist, "Explicit local exchange-correlation potentials,"
0018         % J. Phys. C: Solid State Phys., vol. 4, no. 14, p. 2064, Oct. 1971,
0019         % doi: 10.1088/0022-3719/4/14/022.
0020         A = 0.0368;
0021         B = 21.0;
0022     case 2
0023         % O. Gunnarsson and B. I. Lundqvist, "Exchange and correlation in atoms, molecules, and
0024         % solids by the spin-density-functional formalism,"
0025         % Phys. Rev. B, vol. 13, no. 10, pp. 4274–4298, May 1976,
0026         % doi: 10.1103/PhysRevB.13.4274
0027         A = 0.0545;
0028         B = 11.4;
0029 end
0030 
0031 % check that inputs are non-empty and of equal size (or scalars)
0032 len = [numel(perm),numel(mass),numel(rho)];
0033 doNumeric = all(len~=0) && all(len==1 | len==max(len));
0034 
0035 if doNumeric
0036     % The quantity rs is the radius of the sphere (in bohr) that contains exactly one electron.
0037     % As the density can be zero-valued, we should avoid taking its inverse. Therefore we replace
0038     % all instances of 1/rs with inv_rs, which is proportional to the cube root of the density.
0039     inv_rs = (4*pi/3 * rho).^(1/3);
0040     
0041     % unitless scaling factor
0042     alpha = (4/(9*pi))^(1/3);
0043 
0044     % return the explicit potential (in Hartree)
0045     out = -2/(pi*alpha) * (inv_rs + A*log(1 + B*inv_rs)) .* (mass./(perm.^2));
0046 else
0047     % unitless scaling factor
0048     alpha = (4/(9*pi))^(1/3);
0049 
0050     % return the potential generating function (in Hartree)
0051     out = @(mass,perm,rho) -2/(pi*alpha) .* (mass./(perm.^2)) .* ...
0052         ( (4*pi/3*rho).^(1/3) + A*log( 1 + B*((4*pi/3*rho).^(1/3)) ) );
0053 end
0054 end
